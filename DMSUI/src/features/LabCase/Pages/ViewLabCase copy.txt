/* =====================================================
   FORMAT DATE
===================================================== */

const formatDate = (date) => {
  if (!date) return "ثبت نشده";

  try {
    return new Date(date).toLocaleDateString("fa-IR");
  } catch {
    return date;
  }
};

/* =====================================================
   FORMAT PRICE
===================================================== */

const formatPrice = (price) => {
  if (price === null || price === undefined) return "0";

  return Number(price).toLocaleString("fa-IR");
};

/* =====================================================
   STATUS CONFIG
===================================================== */

const getStatusStyle = (status) => {
  switch (status?.toLowerCase()) {
    case "completed":
      return {
        text: "تکمیل شده",
        icon: CheckCircle2,
        badge:
          "border-blue-200 bg-gradient-to-l from-blue-50 to-indigo-50 text-blue-700",
      };

    case "received":
      return {
        text: "دریافت شده",
        icon: Download,
        badge:
          "border-indigo-200 bg-indigo-50 text-indigo-700",
      };

    case "processing":
      return {
        text: "در حال اجرا",
        icon: Activity,
        badge:
          "border-blue-200 bg-gradient-to-l from-white to-blue-50 text-indigo-700",
      };

    case "pending":
    default:
      return {
        text: status || "در انتظار",
        icon: Clock3,
        badge:
          "border-slate-200 bg-white text-slate-600",
      };
  }
};

/* =====================================================
   STATUS BADGE
===================================================== */

const StatusBadge = ({ status }) => {
  const config = getStatusStyle(status);
  const Icon = config.icon;

  return (
    <span
      className={`inline-flex items-center gap-2 rounded-full border px-3.5 py-2 text-xs font-bold shadow-sm ${config.badge}`}
    >
      <Icon size={15} />

      {config.text}
    </span>
  );
};

/* =====================================================
   SMALL INFO ITEM

      </div>

      <div className="min-w-0">

        <p className="text-[11px] font-semibold text-slate-400">
          {label}

        <p className="mt-1 truncate text-sm font-bold text-slate-700">
          {value || "ثبت نشده"}

    </div>

};
